% image-net post, figure 5 (after d2l.ai, "resnet-block"): the ResNet residual
% block. Left: identity shortcut, x is added straight back. Right: the shortcut
% passes through a 1x1 conv so x can change channels/resolution before the add.
% Colour key shared across the post: conv fa, normalisation/activation fe.
\documentclass[tikz,border=3pt]{standalone}
\input{FIGKIT/preamble.tex}
\begin{document}
\begin{tikzpicture}[fig,
    blk/.style={box=#1, minimum width=26mm, minimum height=6.5mm, inner sep=2pt, font=\rmfamily\small},
    bn/.style={frame=fe, minimum width=26mm, minimum height=6.5mm, inner sep=2pt, font=\rmfamily\small},
    up/.style={arr, shorten <=0pt},
  ]
  % one stack of the residual mapping, bottom to top, starting at #1
  \newcommand{\stack}[2]{%
    \node[blk=fa] (#2c1) at (#1) {$3\times3$ conv};
    \node[bn, above=4mm of #2c1] (#2b1) {batch norm};
    \node[blk=fe, above=4mm of #2b1] (#2r1) {ReLU};
    \node[blk=fa, above=4mm of #2r1] (#2c2) {$3\times3$ conv};
    \node[bn, above=4mm of #2c2] (#2b2) {batch norm};
    \node[op=fa, above=6mm of #2b2] (#2add) {$+$};
    \node[blk=fe, above=5mm of #2add] (#2r2) {ReLU};
    \node[below=12mm of #2c1] (#2x) {$\mathbf{x}$};
    \draw[up] (#2x) -- (#2c1);
    \foreach \a/\b in {c1/b1, b1/r1, r1/c2, c2/b2, b2/add, add/r2}
      \draw[up] (#2\a) -- (#2\b);
    \draw[up] (#2r2.north) -- ++(0,5mm);
    \begin{scope}[on background layer]
      \node[group, inner sep=3.5mm, fit=(#2c1)(#2b2)] (#2grp) {};
    \end{scope}
  }
  % ---- left: identity shortcut
  \stack{0,0}{L}
  \coordinate (Lsplit) at ($(Lx.north)!.45!(Lc1.south)$);
  \draw[arr, rounded corners=3pt] (Lsplit) -| ($(Lgrp.east)+(5mm,0)$) |- (Ladd.east);
  \fill (Lsplit) circle (1.1pt);
  % ---- right: 1x1 conv shortcut
  \stack{58mm,0}{R}
  \coordinate (Rsplit) at ($(Rx.north)!.45!(Rc1.south)$);
  \node[blk=fa, minimum width=20mm, anchor=west] (Rsc) at ($(Rgrp.east |- Rr1)+(6mm,0)$) {$1\times1$ conv};
  \draw[arr, rounded corners=3pt] (Rsplit) -| (Rsc.south);
  \draw[arr, rounded corners=3pt] (Rsc.north) |- (Radd.east);
  \fill (Rsplit) circle (1.1pt);
  % ---- captions
  \node[lab, anchor=north] at ($(Lx.south)+(0,-1mm)$) {identity shortcut};
  \node[lab, anchor=north] at ($(Rx.south)+(10mm,-1mm)$) {1x1 conv shortcut};
\end{tikzpicture}
\end{document}
